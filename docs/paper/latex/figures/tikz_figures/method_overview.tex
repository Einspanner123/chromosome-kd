\begin{minipage}[c]{0.27\textwidth}
\centering
\begin{tikzpicture}[x=0.85in, y=0.85in]
  % --- Panel (a): Trajectories ---
  % RF straight line: noise (0,0) -> GT (1,1)
  \draw[C_RF, line width=1.2pt] (0,0) -- (1,1);

  % DDPM curved line: same endpoints, sinusoidal perpendicular bend.
  % bend = 0.28 * sin(pi*t)^1.5, perp = (-0.707, 0.707)
  % ddpm_x = (1-t) - 0.198 * sin(pi*t)^1.5
  % ddpm_y = (1-t) + 0.198 * sin(pi*t)^1.5
  \draw[C_DDPM, line width=1.2pt, dashed,
        domain=0:1, samples=60, smooth]
    plot ({1-\x-0.198*pow(sin(deg(pi*\x)),1.5)},
          {1-\x+0.198*pow(sin(deg(pi*\x)),1.5)});

  % Noise marker at (0,0): large gray circle
  \filldraw[C_SOURCE, draw=black, line width=0.5pt] (0,0) circle (4pt);
  % GT marker at (1,1): large black circle
  \filldraw[black] (1,1) circle (4pt);

  % Step nodes on RF line at t = 0, 0.33, 0.67, 1.0
  % (position p = (1-t, 1-t))
  \foreach \t in {0, 0.33, 0.67, 1.0} {
    \pgfmathsetmacro{\px}{1-\t}
    \filldraw[C_RF, draw=white, line width=0.4pt] (\px, \px) circle (2pt);
  }

  % Labels for endpoints (descriptors in caption to save space)
  \node[anchor=north east, font=\footnotesize, inner sep=2pt]
    at (-0.02,-0.02) {$\mathbf{x}_1$};
  \node[anchor=south west, font=\footnotesize, inner sep=2pt]
    at (1.02,1.02) {$\mathbf{x}_0$};

  % "decreasing t" arrow below the trajectories
  \draw[->, line width=0.8pt] (0.12,-0.22) -- (0.88,-0.22);
  \node[font=\scriptsize] at (0.5,-0.32) {decreasing $t$ (few-step)};

  % Legend (top-left)
  \node[anchor=north west, font=\scriptsize, inner sep=1pt, align=left]
    at (-0.15, 1.20) {%
    {\color{C_RF}\rule[2pt]{8pt}{1.2pt}}~RF (straight)\\[1pt]
    {\color{C_DDPM}\rule[2pt]{8pt}{1.2pt}}~DDPM (curved)%
  };
\end{tikzpicture}
